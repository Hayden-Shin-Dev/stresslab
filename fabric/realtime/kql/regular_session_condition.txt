Fabric 파이프라인 regular_session의 If 조건 (평일 09:01~15:21 KST에만 활동 실행)

@and(
  and(
    greaterOrEquals(dayOfWeek(addHours(utcnow(), 9)), 1),
    lessOrEquals(dayOfWeek(addHours(utcnow(), 9)), 5)
  ),
  and(
    greaterOrEquals(int(formatDateTime(addHours(utcnow(), 9), 'HHmm')), 901),
    lessOrEquals(int(formatDateTime(addHours(utcnow(), 9), 'HHmm')), 1521)
  )
)
